`timescale 1ns / 1ps

module integrator_new #(parameter WIDTH = 16)(
    input clk, rst, en, detection,
    input [WIDTH-1:0] output_old,
    input [13:0] decay,
    input [WIDTH-1:0] stimolo,
    input [WIDTH-1:0] threshold,
    
    output valid,
    output spike,
    output [WIDTH-1:0] output_new
);

localparam P_SIZE = WIDTH + 13;

reg signed [WIDTH-1:0] p_shift;
reg signed [WIDTH-1:0] r_stimolo[2:0];
reg signed [WIDTH-1:0] r_threshold [3:0];
reg signed [WIDTH-1:0] comparator_in;
reg [3:0] en_shift;

reg [P_SIZE-1:0] supporto_1, supporto_2;

wire signed [15:0] mac_a = output_old[15:0]; // tronca a 16 bit
wire signed [15:0] mac_b = { {2{decay[13]}}, decay }; // segno esteso a 16 bit
wire [31:0] mult_out;

// Istanziazione del blocco DSP
SB_MAC16 #(
    .A_SIGNED(1'b1),
    .B_SIGNED(1'b1),
    .TOPOUTPUT_SELECT(2'b11),
    .BOTOUTPUT_SELECT(2'b11),
    .PIPELINE_16x16_MULT_REG2(1'b1)
) mac_inst (
    .A(mac_a),
    .B(mac_b),
    .C(16'b0),
    .D(16'b0),
    .O(mult_out),
    .CLK(clk),
    .CE(1'b1),
    .IRSTTOP(rst),
    .IRSTBOT(rst),
    .ORSTTOP(1'b0),
    .ORSTBOT(1'b0),
    .AHOLD(1'b0),
    .BHOLD(1'b0),
    .CHOLD(1'b0),
    .DHOLD(1'b0),
    .OHOLDTOP(1'b0),
    .OHOLDBOT(1'b0),
    .OLOADTOP(1'b0),
    .OLOADBOT(1'b0),
    .ADDSUBTOP(1'b0),
    .ADDSUBBOT(1'b0),
    .CI(1'b0),
    .CO(),
    .ACCUMCI(),
    .ACCUMCO(),
    .SIGNEXTIN(),
    .SIGNEXTOUT()
);

// Pipeline
integer i;
always @(posedge clk) begin
    if (rst) begin
        en_shift <= 0;
        p_shift <= 0;
        comparator_in <= 0;
        for(i = 0; i < 4; i = i + 1)
            r_threshold[i] <= 0;
        for(i = 0; i < 3; i = i + 1)
            r_stimolo[i] <= 0;
    end else begin
        en_shift[0] <= en;
        r_stimolo[0] <= stimolo;
        r_threshold[0] <= threshold;

        en_shift[1] <= en_shift[0];
        r_stimolo[1] <= r_stimolo[0];
        r_threshold[1] <= r_threshold[0];

        // Shift aritmetico a destra di 12 bit
        en_shift[2] <= en_shift[1];
        if (mult_out[P_SIZE-1]) begin
            supporto_1 = ~mult_out + 1'b1;
            supporto_2 = supporto_1[P_SIZE-1:12];
            p_shift <= -supporto_2;
        end else begin
            p_shift <= mult_out[P_SIZE-1:12];
        end

        r_stimolo[2] <= r_stimolo[1];
        r_threshold[2] <= r_threshold[1];

        en_shift[3] <= en_shift[2];
        comparator_in <= p_shift + r_stimolo[2];
        r_threshold[3] <= r_threshold[2];
    end
end

assign spike = (comparator_in >= r_threshold[3]) & detection;
assign output_new = spike ? 0 : comparator_in;
assign valid = en_shift[3];

endmodule
